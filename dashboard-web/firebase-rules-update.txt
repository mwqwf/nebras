  Nebras Dashboard — Firebase Rules Update
════════════════════════════════════════════════════════════════════════════════

هذا الملف يحوي القواعد الكاملة لـ Realtime Database و Cloud Storage التي يجب
تطبيقها على مشروع Nebras الأساسيّ (nebras-9118c) لكي يعمل:

  1) سكريبت الجلب من مكتبة نور (Noor Library Fetcher)
  2) نظام رفع الملفّات الموحَّد (Smart Router → Storage Writer)
  3) كتابة سجلات `dashboard_uploads/{fileId}` و `content_unified/files/{fileId}`
  4) قراءة الأقسام `sections_unified/*` من العملاء (مدير/مشرف)

نموذج الأدوار المعتمد في hooks.server.js:
  • role === 'owner'        → صلاحيّات كاملة (CRUD على كلّ شيء)
  • role === 'supervisor'   → كتابة على المحتوى/الأقسام
  • role === 'admin'        → يُترجَم تلقائيّاً إلى 'supervisor'
  • أي شيء آخر / محظور      → قراءة فقط حسب القاعدة

ملاحظة معماريّة مهمّة:
  جميع نقاط /api/admin/* و /api/mshcat/* و /api/oldapp/* تمرّ عبر Admin SDK
  من جانب الخادم، وهو يتجاوز هذه القواعد. القواعد أدناه تحمي:
    • الكتابات المباشرة من المتصفّح عبر Web SDK (Nebras فقط).
    • القراءات المباشرة من تطبيق الموبايل/المتصفّح.
    • أيّ محاولة كتابة من عميل غير مصرّح (Defense-in-depth).

────────────────────────────────────────────────────────────────────────────────
  2) Cloud Storage Rules
  المسار في Firebase Console: Storage → Rules
  Bucket: nebras-9118c.firebasestorage.app
────────────────────────────────────────────────────────────────────────────────

rules_version = '2';

service firebase.storage {
  match /b/{bucket}/o {

    // ── دوال مساعدة ──────────────────────────────────────────
    function isSignedIn() {
      return request.auth != null;
    }

    function userRole() {
      return firestore.get(
        /databases/(default)/documents/dashboard_users/$(request.auth.uid)
      ).data.role;
    }

    // ملاحظة: قواعد Storage لا تستطيع قراءة Realtime Database مباشرةً.
    // الحلّ المعتمد: نسجّل دور كلّ مستخدم في Firebase Auth Custom Claims
    // عند ترقيته إلى supervisor/owner، فيصبح الدور متاحاً عبر:
    //   request.auth.token.role
    // إن لم تكن قد فعَّلت Custom Claims بعد، استخدم الفرع البديل في الأسفل.

    function hasOwnerOrSupervisor() {
      return isSignedIn() && (
        request.auth.token.role == 'owner' ||
        request.auth.token.role == 'supervisor' ||
        // ⚠️ Fallback ضروري لتعرّف اللوحة على المالك إذا لم تنتشر Custom
        // Claims بعد. ضع هنا بريد المالك الفعليّ من القاعدة التشغيليّة
        // (يُقرَأ من `storage.rules` التشغيليّ — لا تطبع البريد علناً).
        request.auth.token.email == '<OWNER_EMAIL_PLACEHOLDER>'
      );
    }

    function isValidUpload() {
      // 200 ميغا حدّ أقصى لكل ملف (مرَن — عدّل حسب الحاجة).
      return request.resource.size < 200 * 1024 * 1024
        && request.resource.contentType.matches('(image|video|audio|application)/.*');
    }

    // ── محتوى لوحة التحكّم (الملفّات الفعلية + الـ thumbnails) ─
    // المسار الذي يكتب فيه storageUpload.js → firebaseUploadContentFile
    // و uploadContentThumbnail. القراءة عامّة (ليتمكّن تطبيق الموبايل من
    // عرض المحتوى)؛ الكتابة لـ owner/supervisor فقط.
    match /dashboard/content/{fileId}/{allPaths=**} {
      allow read: if true;
      allow write: if hasOwnerOrSupervisor() && isValidUpload();
    }

    // ── الرفع العام المباشر إلى Nebras (من smartUpload.js) ────
    match /dashboard/uploads/{allPaths=**} {
      allow read: if true;
      allow write: if hasOwnerOrSupervisor() && isValidUpload();
    }

    // ── thumbnails الأقسام (sections/*) ──────────────────────
    // يكتب فيه moderator.js → uploadSectionThumbnail عند المسارات
    // sections/main/{id}/... و sections/sub/{id}/... وهكذا.
    match /sections/{level}/{sectionId}/{allPaths=**} {
      allow read: if true;
      allow write: if hasOwnerOrSupervisor()
        && request.resource.size < 10 * 1024 * 1024
        && request.resource.contentType.matches('image/.*');
    }

    // ── ملفات سكريبت Noor Library Fetcher ─────────────────────
    // المسار المخصَّص لمحتوى مكتبة نور المُجلَب آلياً.
    match /dashboard/noor-library/{bookId}/{allPaths=**} {
      allow read: if true;
      allow write: if hasOwnerOrSupervisor() && isValidUpload();
    }

    // ── منع كل ما عدا ذلك ─────────────────────────────────────
    match /{path=**} {
      allow read: if false;
      allow write: if false;
    }
  }
}


────────────────────────────────────────────────────────────────────────────────
  3) (اختياري ولكن مُوصى به) — تفعيل Custom Claims للأدوار
────────────────────────────────────────────────────────────────────────────────

قواعد Storage أعلاه تعتمد على `request.auth.token.role` (Custom Claim). يجب
ترقية المستخدمين لمرّة واحدة عبر سكريبت Admin SDK لمنحهم الـ claim المناسب،
وإلّا ستفشل الكتابة لكل من ليس "المالك الرسميّ" بالبريد.

كود الترقية (يُنفَّذ مرّة واحدة في Node.js مع Admin SDK):

  // upgrade-roles.mjs — شغّله من جذر dashboard/
  import { initializeApp, cert } from 'firebase-admin/app';
  import { getAuth } from 'firebase-admin/auth';
  import { getDatabase } from 'firebase-admin/database';
  import { readFileSync } from 'node:fs';

  const sa = JSON.parse(readFileSync('./secrets/serviceAccountKey.json', 'utf8'));
  initializeApp({
    credential: cert(sa),
    databaseURL: 'https://nebras-9118c-default-rtdb.firebaseio.com'
  });

  const usersSnap = await getDatabase().ref('dashboard_users').get();
  if (!usersSnap.exists()) process.exit(0);
  for (const [uid, data] of Object.entries(usersSnap.val() || {})) {
    const role = data.role === 'admin' ? 'supervisor' : (data.role || 'supervisor');
    await getAuth().setCustomUserClaims(uid, { role });
    console.log(`✓ ${uid} → role=${role}`);
  }

ملاحظة: يجب على المستخدم تسجيل الخروج/الدخول مرّة واحدة بعد ترقية الـ claim
لكي يُحدَّث الـ ID Token الذي يحمله متصفّحه.


────────────────────────────────────────────────────────────────────────────────
  4) قواعد المشاريع الثانويّة (Mshcat / OldApp) — مرجعيّة فقط
────────────────────────────────────────────────────────────────────────────────

كلّ كتابة على مشروعَيْ Mshcat و OldApp تمرّ عبر Admin SDK من جانب الخادم
(انظر uploadBridge.js و /api/mshcat/* و /api/oldapp/*)، فلا حاجة لتعديل
قواعد العميل في تلك المشاريع. القراءة العامّة عبر Web SDK تكفي حالياً
وهي مفعّلة افتراضيّاً في القواعد الموجودة هناك.

إن أردت إحكام الحماية مستقبلاً، طبّق نفس مبدأ "قراءة عامّة + كتابة محظورة
من العميل" في Firestore Rules لكلا المشروعين:

  rules_version = '2';
  service cloud.firestore {
    match /databases/{database}/documents {
      match /{document=**} {
        allow read: if true;
        allow write: if false;  // الكتابة عبر Admin SDK فقط
      }
    }
  }
